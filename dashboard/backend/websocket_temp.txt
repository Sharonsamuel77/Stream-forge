# --------------------------------------------------
# WebSocket Metrics
# --------------------------------------------------

@app.websocket("/ws/metrics")
async def websocket_metrics(websocket: WebSocket):

    await websocket.accept()

    previous_processed = None
    previous_time = None

    try:
        while True:

            try:
                response = requests.get(
                    "http://localhost:9000/metrics",
                    timeout=5
                )

                response.raise_for_status()

                text = response.text

                processed = re.search(
                    r"streamforge_messages_processed_total\s+([0-9.]+)",
                    text
                )

                processed_value = (
                    float(processed.group(1))
                    if processed
                    else 0
                )

                current_time = time.time()

                throughput = 0

                if (
                    previous_processed is not None
                    and previous_time is not None
                ):

                    elapsed = current_time - previous_time
                    difference = processed_value - previous_processed

                    if elapsed > 0 and difference >= 0:
                        throughput = difference / elapsed

                previous_processed = processed_value
                previous_time = current_time

                await websocket.send_json({
                    "throughput": round(throughput, 2),
                    "total_events_sec": round(throughput, 2)
                })

            except Exception as e:

                print("WebSocket metrics error:", e)

                await websocket.send_json({
                    "throughput": 0,
                    "total_events_sec": 0
                })

            await asyncio.sleep(2)

    except Exception:
        pass
